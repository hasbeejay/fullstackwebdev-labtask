Problems are pushed onto `errors`.function convertNumber(rawValue, label, errors) {  // Guard clause: undefined, null, '' and '   ' all count as "missing"  if (rawValue === undefined || rawValue === null || String(rawValue).trim() === '') {    errors.push(`${label} is missing`);    return null;  }  const converted = Number(String(rawValue).trim());  if (!Number.isFinite(converted)) {            // catches NaN and Infinity    errors.push(`${label} is not a valid number ("${rawValue}")`);    return null;  }  return converted;} function checkRange(value, label, min, max, errors) {  if (value === null) return;                   // already reported as missing / invalid  if (value < min || value > max) {    errors.push(`${label} must be between ${min} and ${max} (got ${value})`);  }} // switch statement: only the literal strings 'true' / 'false' are acceptedfunction convertBoolean(rawValue, label, errors) {  switch (String(rawValue).trim().toLowerCase()) {    case 'true':      return true;    case 'false':      return false;    default:      errors.push(`${label} must be "true" or "false"`);      return null;  }} function convertProgram(rawValue, errors) {  const program = String(rawValue ?? '').trim().toUpperCase();  if (program === '') {    errors.push('program is missing');    return null;  }  if (!SUPPORTED_PROGRAMS.includes(program)) {    errors.push(`program "${program}" is not supported`);    return null;  }  return program;} // ---------- business rules ---------- function calculateMerit(matric, intermediate, entryTest) {  return round2(    matric * WEIGHTS.matric + intermediate * WEIGHTS.intermediate + entryTest * WEIGHTS.entryTest  );} function decideStatus(merit, intermediate, entryTest, feePaid) {  const eligible =    merit >= RULES.minMerit && entryTest >= RULES.minEntryTest && intermediate >= RULES.minIntermediate;   if (!eligible) return 'REJECTED';  // ternary: strong merit AND paid fee -> SELECTED, otherwise the applicant is held on the waiting list  return merit >= RULES.selectedMerit && feePaid ? 'SELECTED' : 'WAITING';} // Validates one applicant and returns a NEW result object (the source is never touched).export function processApplicant(applicant) {  // Guard clause: the record itself might be null, a number, an array ...  if (typeof applicant !== 'object' || applicant === null || Array.isArray(applicant)) {    return {      id: null, name: null, program: null, valid: false, merit: null,      status: 'INVALID DATA', errors: ['record is not an object']    };  }   const errors = [];   // truthy / falsy: !value is true for undefined, null and ''  if (!applicant.id) errors.push('id is missing');  if (!applicant.name || String(applicant.name).trim() === '') errors.push('name is missing');   const age = convertNumber(applicant.age, 'age', errors);  const matric = convertNumber(applicant.matricMarks, 'matricMarks', errors);  const intermediate = convertNumber(applicant.intermediateMarks, 'intermediateMarks', errors);  const entryTest = convertNumber(applicant.entryTest, 'entryTest', errors);   checkRange(age, 'age', 17, 30, errors);  checkRange(matric, 'matricMarks', 0, 100, errors);  checkRange(intermediate, 'intermediateMarks', 0, 100, errors);  checkRange(entryTest, 'entryTest', 0, 100, errors);   const program = convertProgram(applicant.program, errors);  const feePaid = convertBoolean(applicant.feePaid, 'feePaid', errors);   const result = {    id: applicant.id ?? null,    name: applicant.name ?? null,    program: program ?? String(applicant.program ?? '').trim().toUpperCase(),    valid: errors.length === 0,    merit: null,    status: 'INVALID DATA',    errors  };   // Guard clause: merit is calculated only AFTER the record is known to be valid  if (!result.valid) return result;   result.merit = calculateMerit(matric, intermediate, entryTest);  result.status = decideStatus(result.merit, intermediate, entryTest, feePaid);  return result;} export function processApplicants(applicants) {  const results = [];  for (const applicant of applicants) {    results.push(processApplicant(applicant));  }  return results;} export function summarize(results) {  let valid = 0;  let invalid = 0;  let totalMerit = 0;  let highest = null;  let lowest = null;  const programs = new Map();   for (const result of results) {    const key = result.program === '' ? 'MISSING' : result.program ?? 'MISSING';    programs.set(key, (programs.get(key) ?? 0) + 1);     if (!result.valid) {      invalid++;      continue;    }    valid++;    totalMerit += result.merit;    if (highest === null || result.merit > highest.merit) highest = result;    if (lowest === null || result.merit < lowest.merit) lowest = result;  }   return {    validApplicants: valid,    invalidApplicants: invalid,    highestMerit: highest === null ? null : { id: highest.id, name: highest.name, merit: highest.merit },    lowestMerit: lowest === null ? null : { id: lowest.id, name: lowest.name, merit: lowest.merit },    averageMerit: valid === 0 ? null : round2(totalMerit / valid),    applicantsPerProgram: Object.fromEntries(programs)  };} // 01 / NOTES

Requirements & explanation

Requirements covered

Explicit conversion with Number(), with missing values and NaN detected
Marks between 0–100, age between 17–30, program in BSCS / BSSE / BSAI
Validity decided before merit: Matric 20% + Intermediate 30% + Entry Test 50%
Eligible only if merit ≥ 70, entry test ≥ 60 and intermediate ≥ 60
Status: SELECTED, WAITING, REJECTED or INVALID DATA
Summary: valid / invalid counts, highest, lowest, average and per-program counts
Source data frozen with Object.freeze and proven unchanged

SELECTED vs WAITING

An eligible applicant is SELECTED when merit is ≥ 75 and the fee is paid. Everyone else who is eligible is WAITING: a borderline merit of 70–74.99, or a strong applicant whose fee is still unpaid.

The reasoning: a seat cannot be confirmed without the fee, and borderline candidates are kept in reserve in case seats open up. Not eligible means REJECTED; broken data means INVALID DATA.

Why empty strings are checked first

Number('') is 0, not NaN, so a blank Intermediate mark would silently become a perfect-looking zero. A guard clause catches undefined, null and blank strings before conversion, and only then is Number() applied.

Marks are compared with strict checks, not truthiness, because 0 is a legitimate mark. Truthiness is used only for id and name, where an empty value really does mean missing.

Test data: normal, boundary, invalid

  • Normal: the three records from the case study.
  • Boundary: merit exactly 70, entry test 59, intermediate 59, age 17 and 30, perfect scores, lowercase program name.
  • Invalid: age 16, marks 101, unsupported program, bad fee flag, a record that is wrong in every field, and null.

Guard clauses keep the flow flat

processApplicant() returns early for a non-object record and again for any record with validation errors. The merit and status code at the bottom therefore only ever sees clean numbers, and no NaN can leak into the summary.
