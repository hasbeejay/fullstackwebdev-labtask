// Student.js - default export: the Student classimport { calculateAverage, calculateGPA, calculateGrade } from './gradeUtils.js'; export default class Student {  constructor(id, name, courses = []) {    this.id = id;    this.name = name;    this.courses = courses;          // [{ code, marks }]  }   get average() {    return calculateAverage(this.courses.map((course) => course.marks));  }   get gpa() {    return calculateGPA(this.courses.map((course) => course.marks));  }   // A plain, JSON-friendly snapshot of the student  toReport() {    return {      id: this.id,      name: this.name,      courses: this.courses.map((course) => ({ ...course, grade: calculateGrade(course.marks) })),      average: this.average,      gpa: this.gpa    };  }}
task5/gradeUtils.jsJavaScript · ES moduleDownload
// gradeUtils.js - named exports only const GRADE_POLICY = [  { min: 85, grade: 'A', points: 4.0 },  { min: 80, grade: 'A-', points: 3.67 },  { min: 75, grade: 'B+', points: 3.33 },  { min: 70, grade: 'B', points: 3.0 },  { min: 65, grade: 'B-', points: 2.67 },  { min: 60, grade: 'C+', points: 2.33 },  { min: 50, grade: 'C', points: 2.0 },  { min: 0, grade: 'F', points: 0.0 }]; const round2 = (value) => Math.round(value * 100) / 100; function findBand(marks) {  if (typeof marks !== 'number' || Number.isNaN(marks)) {    throw new TypeError(`marks must be a number, received ${typeof marks}`);  }  if (marks < 0 || marks > 100) {    throw new RangeError(`marks must be between 0 and 100, received ${marks}`);  }  return GRADE_POLICY.find((band) => marks >= band.min);} export function calculateGrade(marks) {  return findBand(marks).grade;} export function calculateGradePoints(marks) {  return findBand(marks).points;} export function calculateAverage(values) {  if (!Array.isArray(values)) throw new TypeError('calculateAverage expects an array');  if (values.length === 0) return 0;  return round2(values.reduce((sum, value) => sum + value, 0) / values.length);} export function calculateGPA(marksList) {  return calculateAverage(marksList.map((marks) => calculateGradePoints(marks)));}

// transcriptService.js - parsing, validation and report generationimport Student from './Student.js';                                   // default importimport { calculateAverage as averageOf } from './gradeUtils.js';      // named import with an alias function isObject(value) {  return typeof value === 'object' && value !== null && !Array.isArray(value);} // 1. Parse JSON safely: SyntaxError is converted into a clearer, domain-level Errorexport function parseTranscript(jsonText) {  if (typeof jsonText !== 'string') {    throw new TypeError(`Transcript data must be a JSON string, received ${typeof jsonText}`);  }  try {    return JSON.parse(jsonText);  } catch (error) {    throw new Error(`Malformed JSON: ${error.message}`, { cause: error });  }} // 2. Validate the parsed data and build Student objectsexport function buildStudents(data) {  if (!Array.isArray(data)) {    throw new TypeError('Transcript root must be an array of students');  }   const seenIds = new Set();   return data.map((entry, index) => {    const where = `student at index ${index}`;     if (!isObject(entry)) throw new TypeError(`${where} must be an object`);     const { id, name, courses } = entry;     if (id === undefined || id === null || id === '') {      throw new TypeError(`${where} has a missing id`);    }    if (seenIds.has(id)) {      throw new Error(`Duplicate student id: ${id}`);    }    seenIds.add(id);     if (typeof name !== 'string' || name.trim() === '') {      throw new TypeError(`student ${id} has a missing name`);    }    if (!Array.isArray(courses)) {      throw new TypeError(`student ${id} must have a courses array`);    }     courses.forEach((course, courseIndex) => {      const courseWhere = `student ${id}, course at index ${courseIndex}`;       if (!isObject(course)) throw new TypeError(`${courseWhere} must be an object`);      if (typeof course.code !== 'string' || course.code.trim() === '') {        throw new TypeError(`${courseWhere} has a missing course code`);      }      if (typeof course.marks !== 'number' || Number.isNaN(course.marks)) {        throw new TypeError(`${courseWhere} (${course.code}) marks must be a number`);      }      if (course.marks < 0 || course.marks > 100) {        throw new RangeError(`${courseWhere} (${course.code}) marks ${course.marks} are outside 0-100`);      }    });     return new Student(id, name.trim(), courses);  });} // 3. Generate a plain JavaScript report objectexport function buildReport(students) {  const reports = students.map((student) => student.toReport());  return {    totalStudents: reports.length,    classAverage: averageOf(reports.map((report) => report.average)),    students: reports  };} // 4. Convert the final report back to JSONexport function reportToJSON(report) {  return JSON.stringify(report, null, 2);} // Convenience pipeline: JSON text in -> report object outexport function processTranscript(jsonText) {  return buildReport(buildStudents(parseTranscript(jsonText)));}

05 / NOTES

Requirements & explanation

Requirements covered

Four files: main.js, Student.js, gradeUtils.js, transcriptService.js
Student is a default export; grade helpers are named exports
Default import, named imports and an aliased import (calculateAverage as averageOf)
Malformed JSON, missing IDs, duplicate IDs, bad marks and missing course codes all rejected
Average and GPA per student, report object, report converted back to JSON
Caller demonstrates try, catch and finally

Who imports what

FileExportsImports
gradeUtils.jsnamed: calculateGrade, calculateGradePoints, calculateAverage, calculateGPAnothing
Student.jsexport default class Studentnamed imports from gradeUtils
transcriptService.jsnamed: parseTranscript, buildStudents, buildReport, reportToJSON, processTranscriptdefault Student; calculateAverage as averageOf
main.jsnothing (entry point)default Student; named service functions; calculateGrade as gradeOf

Which error for which problem

ProblemError type
Malformed JSON textError (wraps the SyntaxError as its cause)
Missing student id, missing course code, wrong shape or typeTypeError
Duplicate student idError
Marks below 0 or above 100RangeError

A RangeError means "the type is right but the value is outside the allowed range", while a TypeError means the value is not the kind of thing expected. The caller can react differently to each.

try / catch / finally

Every test case in main.js runs inside try. A failure is caught and printed as name: message, and the finally block runs after both success and failure, which is where cleanup (closing a file, releasing a lock) belongs. The output shows [finally] after every case.
