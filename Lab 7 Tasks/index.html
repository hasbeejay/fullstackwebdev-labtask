// Lab 7 - Part 2: Persistent Store using atomic file writes
// Atomic temp-file + rename, recursive mkdir, serialized write queue
const fs = require('node:fs/promises');
const path = require('node:path');
const { randomUUID } = require('node:crypto');

function createStore(file, initialData = []) {
  let writeQueue = Promise.resolve();

  async function loadFromFile() {
    try {
      return JSON.parse(await fs.readFile(file, 'utf8'));
    } catch (error) {
      if (error.code === 'ENOENT') {
        await fs.mkdir(path.dirname(file), { recursive: true });
        await fs.writeFile(file, JSON.stringify(initialData, null, 2), 'utf8');
        return initialData;
      }
      throw error;
    }
  }

  function edit(change) {
    const operation = writeQueue.then(async () => {
      const current = await loadFromFile();
      const result = await change(current);
      if (result && result.changed) {
        const toSave = result.data !== undefined ? result.data : current;
        await fs.mkdir(path.dirname(file), { recursive: true });
        const temp = `${file}.${randomUUID()}.tmp`;
        try {
          await fs.writeFile(temp, JSON.stringify(toSave, null, 2), 'utf8');
          await fs.rename(temp, file);
        } catch (error) {
          await fs.rm(temp, { force: true }).catch(() => {});
          throw error;
        }
      }
      return result ? result.value : null;
    });
    writeQueue = operation.then(() => undefined, () => undefined);
    return operation;
  }

  return { list, edit, filePath: file };
}

Architectural Comparison · In-Memory vs. Persistent

A comparative evaluation of the two backend architectures implemented in Lab 7.

Attribute Part 1: In-Memory API Part 2: Atomic Persistent File API
Data Durability Transient — data resets whenever the server process restarts Permanent — saved to disk in JSON files (products.json, etc.)
Write Mechanism Direct JS array operations (push, splice) Atomic temp-file write + rename (fs.rename) to prevent corrupt files
Concurrency Control Single-threaded Node event loop Serialized Promise Write Queue preventing interleaved writes
Latency & Throughput Microsecond reads/writes directly in RAM Disk I/O latency on writes; cached in-memory list operations
Ideal Use Cases Fast caches, session tokens, unit testing, transient state Production catalogs, shopping carts, persistent order databases
